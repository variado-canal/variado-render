import numpy as np
import sfx
from sfx import SR, sine, env, bp_noise, rng
def lp(x,fc):
    X=np.fft.rfft(x);f=np.fft.rfftfreq(len(x),1/SR);X*=1/(1+(f/fc)**4);return np.fft.irfft(X,len(x))
def hp(x,fc):
    X=np.fft.rfft(x);f=np.fft.rfftfreq(len(x),1/SR);X*=(f/fc)**4/(1+(f/fc)**4);return np.fft.irfft(X,len(x))
def fade(x,a=.25,b=.35):
    n=len(x);t=np.arange(n)/SR;return x*np.minimum(1,t/a)*np.minimum(1,(n/SR-t)/b)
def pink(n,seed):
    r=np.random.RandomState(seed);x=r.randn(n);X=np.fft.rfft(x);f=np.fft.rfftfreq(n,1/SR);X/=np.sqrt(np.maximum(f,30)/30);y=np.fft.irfft(X,n);return y/np.abs(y).max()
def amb(kind,ln):
    n=int(SR*ln);t=np.arange(n)/SR
    if kind=='cuarto':   # cuarto en calma: aire de habitación + zumbido bajo
        x=lp(pink(n,11),700)*.05+.012*sine(50,n)*(1+.3*np.sin(t*.7))
    elif kind=='cerebro': # dentro del cerebro: zumbido profundo y latido suave
        x=sum(a*sine(f,n)*(1+.35*np.sin(t*w)) for f,a,w in ((55,.03,.4),(82.4,.022,.31),(110,.015,.23)))+hp(pink(n,12),2500)*.012
        for b in np.arange(.2,ln,1.35):
            i=int(b*SR);m=min(int(SR*.4),n-i)
            if m>0: x[i:i+m]+=sine(58,m)*env(m,.01,.18)*.05
    else:                # brillo: aire claro y brillante
        x=bp_noise(n,6000,6000,1.5,rng(13))*.012*(1+.5*np.sin(t*1.1))+sum(a*sine(f,n)*(1+.3*np.sin(t*w)) for f,a,w in ((261.6,.012,.5),(329.6,.01,.37),(392,.009,.29)))
    return fade(x,.4,.5)
def bubble():
    n=int(SR*.35);f=np.linspace(320,780,n)**1.0;x=sine(f,n)*env(n,.004,.12)+.4*sine(f*2,n)*env(n,.004,.06);return x*.5
def fill(ln):
    n=int(SR*ln);t=np.linspace(0,1,n);x=bp_noise(n,500,2600,3,rng(21))*.5+.25*sine(300+700*t,n)*(0.5+.5*np.sin(t*60));return fade(x*.45,.1,.15)
def tictoc(k):
    n=int(SR*.06);f=1800 if k%2==0 else 1350;return (sine(f,n)*env(n,.0004,.03)+.6*np.random.RandomState(k).randn(n)*env(n,.0002,.004))*.55
def yawn():
    n=int(SR*1.5);t=np.linspace(0,1,n);fm=500+700*np.sin(np.pi*t)
    air=bp_noise(n,500,1200,2.2,rng(31))*np.sin(np.pi*np.clip(t,0,1))**1.4
    v=sine(210-100*t,n)*np.sin(np.pi*np.clip(t*1.1,0,1))**2*(1+.05*np.sin(t*60))
    return (air*.5+v*.22)*.9
def zzz(k):
    n=int(SR*.55);f=np.linspace(520,380,n)*(1-.08*k);return sine(f,n)*env(n,.05,.3,3)*(1+.1*np.sin(np.linspace(0,30,n)))*.22
def plip(k):
    r=rng(k+60);n=int(SR*.12);b=r.uniform(600,900);f=np.linspace(b,b*2,n);return sine(f,n)*env(n,.001,.05)*.55
def klik(k):
    r=rng(k+70);n=int(SR*.35);x=np.zeros(n)
    x+=r.randn(n)*env(n,.0002,.004)*.8
    x+=sine(r.uniform(300,360),n)*env(n,.001,.06)*.8
    x+=sine(2400+k*200,n)*env(n,.0004,.14)*.3
    return x*.8
def scan():
    n=int(SR*1.3);t=np.linspace(0,1,n);f=400+1400*t;x=sine(f,n)*np.sin(np.pi*t)**2*.18+bp_noise(n,3000,6000,2,rng(41))*np.sin(np.pi*t)*.08;return x
def steam():
    n=int(SR*1.0);x=hp(np.random.RandomState(51).randn(n),3000);x/=np.abs(x).max();t=np.linspace(0,1,n);return x*np.sin(np.pi*t)**1.5*.1
def sip():
    n=int(SR*.6);t=np.linspace(0,1,n);x=bp_noise(n,500,1700,3,rng(61))*np.sin(np.pi*t)*(.6+.4*np.sin(t*40))*.5
    g=int(SR*.12);i=int(SR*.38);x[i:i+g]+=sine(np.linspace(170,85,g),g)*env(g,.004,.07)*.6;return x
def snap(k):
    n=int(SR*.12);f=1100+130*k;return (sine(f,n)*env(n,.0005,.03)+.6*np.random.RandomState(k+80).randn(n)*env(n,.0002,.004))*.6
def bonk(k):
    n=int(SR*.25);f=np.linspace(280,140,n);return sine(f,n)*env(n,.001,.09)*.8*(1+.0*k)
def burst():
    n=int(SR*1.6);x=np.zeros(n);m=int(SR*.45);t=np.linspace(0,1,m);x[:m]+=bp_noise(m,400,5000,1.6,rng(91))*t**1.6*.6
    x[int(SR*.35):int(SR*.35)+int(SR*.9)]+=sine(58,int(SR*.9))*env(int(SR*.9),.005,.5,3)*.7
    for j,f in enumerate((1568,1976,2349,2637,3136,3951)):
        i=int(SR*(.3+j*.07));m=min(int(SR*.5),n-i);x[i:i+m]+=sine(f,m)*env(m,.002,.22)*.28
    return x
EXTRA={'amb':lambda e:amb(e.get('kind','cuarto'),e.get('len',3)),'bubble':lambda e:bubble(),'fill':lambda e:fill(e.get('len',2)),'tictoc':lambda e:tictoc(int(e.get('k',0))),
 'yawn':lambda e:yawn(),'zzz':lambda e:zzz(int(e.get('k',0))),'plip':lambda e:plip(int(e.get('k',0))),'klik':lambda e:klik(int(e.get('k',0))),'scan':lambda e:scan(),
 'steam':lambda e:steam(),'sip':lambda e:sip(),'snap':lambda e:snap(int(e.get('k',0))),'bonk':lambda e:bonk(int(e.get('k',0))),'burst':lambda e:burst()}
def synth(events,dur):
    N=int(SR*(dur+1.0));L=np.zeros(N);R=np.zeros(N);i=0
    for e in events:
        t=e['type']
        try:
            if t in EXTRA: s=EXTRA[t](e)
            else:
                one=sfx.synth([dict(e,t=0)],.01) if False else None
                s=None
        except Exception as ex: print('sfx2 err',t,ex);continue
        if s is None:
            tmp=sfx.synth([dict(e,t=0)],max(.01,2.0));s=tmp[:int(SR*1.8),0]
        pan=.5 if t=='amb' else .5+.35*np.sin(i*2.399);s0=int(e['t']*SR);m=min(len(s),N-s0)
        if m<=0: continue
        g=1.0 if t=='amb' else 1.4
        L[s0:s0+m]+=s[:m]*(1-pan)*g;R[s0:s0+m]+=s[:m]*pan*g;i+=1
    out=np.stack([L,R],1);out/=max(1e-9,np.abs(out).max())/0.8;return out
