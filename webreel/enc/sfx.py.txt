import numpy as np, random
SR=44100
def rng(seed): return np.random.RandomState(int(seed)%2**31)
def env(n,a=0.002,d=0.2,curve=4.0):
    t=np.arange(n)/SR; e=np.exp(-t*curve/max(d,1e-3)); att=np.minimum(1,t/max(a,1e-4)); return e*att
def bp_noise(n,f0,f1,q=6,r=None):
    # barrido de ruido filtrado (resonador simple por FFT)
    r=r or np.random; x=r.randn(n); X=np.fft.rfft(x); fr=np.fft.rfftfreq(n,1/SR)
    out=np.zeros(n); seg=max(n//24,64)
    for i in range(0,n,seg):
        c=f0+(f1-f0)*i/n; ch=x[i:i+seg]; Xc=np.fft.rfft(ch,seg*2); fc=np.fft.rfftfreq(seg*2,1/SR)
        g=np.exp(-((fc-c)/(c/q))**2); y=np.fft.irfft(Xc*g)[:seg]; out[i:i+len(y)]=y[:min(seg,n-i)]
    return out/(np.abs(out).max()+1e-9)
def sine(f,n,ph=0): return np.sin(2*np.pi*np.cumsum(np.full(n,f) if np.isscalar(f) else f)/SR+ph)
def clink(k,glass):
    r=rng(k*7+1); n=int(SR*(.45 if glass else .6)); base=r.uniform(2000,3400)
    x=np.zeros(n)
    for p,a,dec in ((1,1,.22),(2.76,.6,.14),(5.4,.35,.08),(8.9,.2,.05)):
        x+=a*sine(base*p*r.uniform(.99,1.01),n)*env(n,.0008,dec)
    x+=.5*r.randn(n)*env(n,.0003,.012)
    if glass: x+=.5*sine(r.uniform(820,1050),n)*env(n,.001,.18)   # resonancia de vidrio
    else: x+=.45*sine(r.uniform(300,420),n)*env(n,.001,.07)       # cuerpo de metal al apilar
    return x*r.uniform(.55,.85)
def pop(p):
    n=int(SR*.14); f=np.linspace(700*p,170*p,n); return sine(f,n)*env(n,.001,.07)*.9
def flip():
    r=rng(3); o=np.zeros(int(SR*.55))
    for i,off in enumerate((0,.06,.13,.2)):
        n=int(SR*.07); x=bp_noise(n,5200-i*500,2200,2.5,r)*env(n,.002,.04); s=int(off*SR); o[s:s+n]+=x*(.7-.12*i)
    return o
def whoosh(d,ln,amp):
    n=int(SR*ln); r=rng(int(ln*100)+d); x=bp_noise(n,300 if d>0 else 3200,3200 if d>0 else 300,1.6,r)
    t=np.linspace(0,1,n); e=np.sin(np.pi*np.clip(t,0,1))**1.6; return x*e*amp*.9
def deflate():
    n=int(SR*.45); f=450*np.exp(-np.linspace(0,1.6,n))+120; f=f*(1+.04*np.sin(np.linspace(0,40,n))); return sine(f,n)*env(n,.005,.4,3)*.55
def boing():
    n=int(SR*.55); t=np.linspace(0,1,n); f=180+340*np.exp(-t*7)*np.cos(t*38); x=sine(f,n)*env(n,.002,.4,3)
    x+=.35*sine(f*2.01,n)*env(n,.002,.25,3); return x*.7
def tick(k,soft):
    n=int(SR*.09); r=rng(k+90); f=520*(2**([0,2,4,7,9,12,14,16,19,21,24,26][k%12]/12.))
    return (sine(f,n)*env(n,.001,.05)+.5*r.randn(n)*env(n,.0005,.006))*.7
def count(k):
    n=int(SR*.035); f=900+1700*k; return (sine(f,n)*env(n,.0005,.02)+.35*np.random.RandomState(int(k*100)).randn(n)*env(n,.0003,.004))*.6
def ding():
    n=int(SR*1.4); x=sum(a*sine(880*p,n)*env(n,.002,d) for p,a,d in ((1,1,1.1),(2.01,.5,.7),(2.76,.3,.5),(4.07,.2,.3)));return x*.6
def riser(ln):
    n=int(SR*ln); t=np.linspace(0,1,n); f=220*(1+t*6)**1.2; x=.5*sine(f,n)+.35*bp_noise(n,500,6000,1.2,rng(4)); return x*(t**1.7)*np.minimum(1,(1-t)*12+.05)*.75
def sparkle(k):
    r=rng(k+500); n=int(SR*.3); f=[2093,2637,3136,3520,4186][r.randint(0,5)]*r.choice([1,1,1.5])
    return (sine(f,n)*env(n,.001,.16)+.4*sine(f*2.01,n)*env(n,.001,.08))*.4
def thump():
    n=int(SR*1.6); f=70*np.exp(-np.linspace(0,.5,n))+38; x=sine(f,n)*env(n,.004,1.3,3)+.25*bp_noise(n,120,60,1.5,rng(7))*env(n,.002,.3); return x*.9
def ambient(n,rs=1):
    r=np.random.RandomState(rs); x=r.randn(n); X=np.fft.rfft(x); f=np.fft.rfftfreq(n,1/SR); X*=1/np.sqrt(np.maximum(f,40)/40); y=np.fft.irfft(X,n)
    t=np.arange(n)/SR; y=y/np.abs(y).max()*(.035+.012*np.sin(t*.9)); return y
def synth(events,dur):
    N=int(SR*(dur+1.0)); L=np.zeros(N); R=np.zeros(N); i=0
    amb=ambient(N); L+=amb; R+=amb*.97
    for e in events:
        t=e['type']; k=e.get('k',0)
        try:
            if t=='clink': s=clink(e.get('k',0)+i,e.get('glass',0))
            elif t=='pop': s=pop(e.get('p',1))
            elif t=='flip': s=flip()
            elif t=='whoosh': s=whoosh(e.get('dir',1),e.get('len',.5),e.get('amp',.6))
            elif t=='deflate': s=deflate()
            elif t=='boing': s=boing()
            elif t=='tick': s=tick(int(k),1)
            elif t=='count': s=count(k)
            elif t=='ding': s=ding()
            elif t=='riser': s=riser(e.get('len',1))
            elif t=='sparkle': s=sparkle(int(k))
            elif t=='thump': s=thump()
            else: continue
        except Exception as ex: print('sfx err',t,ex); continue
        pan=.5+.35*np.sin(i*2.399); s0=int(e['t']*SR); m=min(len(s),N-s0)
        L[s0:s0+m]+=s[:m]*(1-pan)*1.4; R[s0:s0+m]+=s[:m]*pan*1.4; i+=1
    out=np.stack([L,R],1); out/=max(1e-9,np.abs(out).max())/0.8; return out
